# NeonArch — reglas de sudo restringidas (Tema 9)
# El grupo wheel NO tiene ALL. Solo comandos concretos, y con log.

# Roy puede gestionar servicios systemd sin password (útil para el proyecto)
roy ALL=(root) NOPASSWD: /usr/bin/systemctl restart sshd, /usr/bin/systemctl status sshd

# Roy puede usar pacman solo para consultas (no instalar)
roy ALL=(root) /usr/bin/pacman -Q, /usr/bin/pacman -Qi, /usr/bin/pacman -Ql

# El resto de comandos del grupo wheel requieren password y quedan logueados
%wheel ALL=(ALL:ALL) ALL
